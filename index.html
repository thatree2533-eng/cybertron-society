""" Cybertron AI — ระบบหลัก + ปลั๊กอิน รูปแบบ: Plugin-Based Architecture เสียบต่อได้ตลอด ไม่ต้องรื้อระบบครับ 🤍✨ """ import json from abc import ABC, abstractmethod from typing import Dict, List, Optional from datetime import datetime # ═══════════════ ฐานข้อมูลปลั๊กอิน ═══════════════ class BasePlugin(ABC): """คลาสแม่ของทุกปลั๊กอิน — ทำตามนี้เสียบได้ทันที""" name: str plugin_id: str description: str status: str = "inactive" @abstractmethod async def execute(self, user_input: str, **kwargs) -> Dict: """รับคำสั่ง → ทำงาน → คืนผลลัพธ์""" pass def get_info(self) -> Dict: return { "id": self.plugin_id, "name": self.name, "description": self.description, "status": self.status } # ═══════════════ ปลั๊กอินแต่ละตัว ═══════════════ class ChatPlugin(BasePlugin): def __init__(self, provider: str = "auto"): self.plugin_id = "chat_general" self.name = "แชตถาม-ตอบทั่วไป" self.description = "เชื่อมต่อ GPT / Claude สลับได้อัตโนมัติ" self.status = "active" self.provider = provider async def execute(self, user_input: str, **kwargs) -> Dict: return { "plugin": self.plugin_id, "provider": self.provider, "reply": f"ได้รับข้อความ: {user_input}", "timestamp": datetime.now().isoformat() } class ImageGenPlugin(BasePlugin): def __init__(self, model: str = "dalle"): self.plugin_id = "image_generator" self.name = "สร้างภาพ & กราฟิก" self.description = "รองรับ Midjourney / DALL-E / Flux" self.status = "active" self.model = model async def execute(self, user_input: str, **kwargs) -> Dict: return { "plugin": self.plugin_id, "model": self.model, "prompt": user_input, "status": "generating", "preview_url": "..." } class KnowledgePlugin(BasePlugin): def __init__(self): self.plugin_id = "knowledge_base" self.name = "คลังความรู้" self.description = "ค้นหาข้อมูลจากฐานความรู้ภายในระบบ" self.status = "active" async def execute(self, user_input: str, **kwargs) -> Dict: return { "plugin": self.plugin_id, "query": user_input, "found": True, "source": "Vector Database" } # ═══════════════ ตัวจัดการกลาง ═══════════════ class AIAgentOrchestrator: """ตัวสลับสมองกลกลาง — เพิ่มปลั๊กอินใหม่เสียบตรงนี้ได้เลย""" def __init__(self): self.plugins: Dict[str, BasePlugin] = {} self._register_core_plugins() def _register_core_plugins(self): """ลงทะเบียนปลั๊กอินหลัก""" self.register(ChatPlugin()) self.register(ImageGenPlugin()) self.register(KnowledgePlugin()) def register(self, plugin: BasePlugin): """เพิ่มปลั๊กอินใหม่ — เรียกใช้ตรงนี้ครับ""" self.plugins[plugin.plugin_id] = plugin print(f"✅ เสียบปลั๊กอิน: {plugin.name} [{plugin.plugin_id}]") def list_plugins(self) -> List[Dict]: """ดูรายการปลั๊กอินทั้งหมด""" return [p.get_info() for p in self.plugins.values()] async def route_request(self, plugin_id: str, user_input: str, **kwargs) -> Dict: """ส่งคำขอไปยังปลั๊กอินที่ถูกต้อง""" if plugin_id not in self.plugins: return { "error": f"ไม่พบปลั๊กอิน: {plugin_id}", "available": list(self.plugins.keys()) } return await self.plugins[plugin_id].execute(user_input, **kwargs) # ═════════════── ตัวอย่างการใช้งาน ═══════════════ if __name__ == "__main__": import asyncio # สร้างระบบหลัก orchestrator = AIAgentOrchestrator() # 📋 ดูรายการปลั๊กอินที่มี print("\n📋 ปลั๊กอินที่เสียบใช้งานอยู่:") for p in orchestrator.list_plugins(): print(f" • {p['id']:20} | {p['name']} | {p['status']}") # ✅ ส่งคำสั่งไปที่แชต async def demo(): result = await orchestrator.route_request( "chat_general", "สวัสดีครับ ระบบทำงานอยู่ไหม" ) print("\n📤 ผลลัพธ์จากแชต:") print(json.dumps(result, ensure_ascii=False, indent=2)) asyncio.run(demo()) # ➕ เพิ่มปลั๊กอินใหม่ — แค่เขียนคลาสแล้ว register() print("\n➕ เพิ่มปลั๊กอินใหม่ได้ตลอดเวลาครับ...")
